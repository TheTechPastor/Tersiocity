  .map((year, index) => year + index);

const month = ("0" + (d.month() + 1)).slice(-2);
const day = ("0" + (d.date())).slice(-2);
const dateString = month + '-' + day;

	var note = dv.page(`${journalPath}/${y}-${dateString}`);
	if (note) {

});
