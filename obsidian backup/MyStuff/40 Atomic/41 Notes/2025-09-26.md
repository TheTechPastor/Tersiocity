## 📆 Last 7 Days
 
```dataviewjs
// Show files created OR modified in the last 7 days, grouped by day
const cutoff = dv.date("today").minus({ days: 7 });

const items = dv.pages("")
  .where(p => (p.file.cday && p.file.cday >= cutoff) || (p.file.mday && p.file.mday >= cutoff))
  .map(p => {
    const dt = (p.file.mday && p.file.cday && p.file.mday > p.file.cday) ? p.file.mday : (p.file.mday ?? p.file.cday);
    return { p, dt };
  });

// Group by ISO date key; display a natural-language header
const groups = {};
for (const { p, dt } of items) {
  if (!dt) continue;
  const iso = dt.toISODate(); // safe for sorting
  const header = dt.toFormat("EEEE, MMMM d"); // e.g., Tuesday, September 3
  if (!groups[iso]) groups[iso] = { header, rows: [] };
  groups[iso].rows.push(p);
}

// Sort days newest → oldest using ISO keys
const days = Object.keys(groups).sort((a, b) => b.localeCompare(a));

for (const day of days) {
  dv.header(3, groups[day].header);
  // Optional: sort rows by last modified within the day
  const rows = groups[day].rows.sort((a, b) => (b.file.mday?.toMillis?.() ?? 0) - (a.file.mday?.toMillis?.() ?? 0));
  dv.table(
    ["File", "Created", "Modified"],
    rows.map(p => [p.file.link, p.file.cday, p.file.mday])
  );
}
```
